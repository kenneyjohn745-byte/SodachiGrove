<div id="msg">loading Sodachi Grove...</div>

<script>

function showMsg(t) { document.getElementById('msg').textContent = t; }

window.addEventListener('error', function(e) { showMsg('JS Error: ' + e.message); });


if (typeof BABYLON === 'undefined') {

    showMsg('Babylon failed to load. Check internet connection.');

} else {

    try {

        const canvas = document.getElementById('renderCanvas');

        const engine = new BABYLON.Engine(canvas, true);


        function createMaterial(name, r, g, b, scene) {

            const mat = new BABYLON.StandardMaterial(name, scene);

            mat.diffuseColor = new BABYLON.Color3(r, g, b);

            return mat;

        }


        const createScene = function() {

            const scene = new BABYLON.Scene(engine);

            scene.clearColor = new BABYLON.Color4(0.3, 0.35, 0.45, 1);

            const sun = new BABYLON.DirectionalLight('sun', new BABYLON.Vector3(0.5, -1, 0.3), scene);

            sun.intensity = 1.0;


            const ground = BABYLON.MeshBuilder.CreateGround('gnd', {width: 24000, height: 24000}, scene);

            ground.material = createMaterial('gndMat', 0.1, 0.12, 0.1, scene);


            const camera = new BABYLON.ArcRotateCamera('camera', -Math.PI / 2, Math.PI / 4, 1800, new BABYLON.Vector3(0, 300, 0), scene);

            camera.attachControl(canvas, true);


            showMsg('Sodachi Grove Foundation loaded.');

            return scene;

        };


        const scene = createScene();

        engine.runRenderLoop(function() { scene.render(); });

    } catch(err) {

        showMsg('3D Error: ' + err.message);

    }

}

</script>
